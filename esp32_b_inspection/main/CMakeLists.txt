idf_component_register(
    SRCS "inspection_station.cpp" "platform.cpp" "../tests/logic_compile_checks.cpp"
    INCLUDE_DIRS "."
    REQUIRES esp_wifi esp_event esp_netif esp_timer nvs_flash lwip
)
target_compile_features(${COMPONENT_LIB} PUBLIC cxx_std_17)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror)
option(MCT431_LOST_ACK_TEST "Suppress the first inspection ACK pair once per boot" OFF)
if(MCT431_LOST_ACK_TEST)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MCT431_LOST_ACK_TEST=1)
endif()
option(MCT431_RETRY_EXHAUSTION_TEST "Withhold all responses for one inspection job" OFF)
if(MCT431_RETRY_EXHAUSTION_TEST)
    if(MCT431_LOST_ACK_TEST)
        message(FATAL_ERROR "Enable only one inspection fault injection")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MCT431_RETRY_EXHAUSTION_TEST=1)
endif()
option(MCT431_DUPLICATE_TEST "Trace inspection execution and replay one selected result" OFF)
if(MCT431_DUPLICATE_TEST)
    if(MCT431_LOST_ACK_TEST OR MCT431_RETRY_EXHAUSTION_TEST)
        message(FATAL_ERROR "Enable only one inspection test mode")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE MCT431_DUPLICATE_TEST=1)
endif()
